 public bool IsMovingBlur { get { return movingBlur != null; } }

 public bool TryStartBlurMove(Point point)
 {
  var mark = FindBlur(point);
  if (mark == null) return false;
  dragOrigin = point;
  originalBlurStart = (Point)markStart.GetValue(mark);
  originalBlurEnd = (Point)markEnd.GetValue(mark);
  dragBlurBounds = BlurBounds(mark, (Rectangle)selection.GetValue(form));
  movingBlur = mark;
  selectedBlur = mark;
  form.Capture = true;
  form.Cursor = Cursors.SizeAll;
  return true;
 }

 public void MoveBlur(Point point)
 {
  var area = (Rectangle)selection.GetValue(form);
  int x = Math.Max(area.Left, Math.Min(area.Right - dragBlurBounds.Width,
   dragBlurBounds.X + point.X - dragOrigin.X));
  int y = Math.Max(area.Top, Math.Min(area.Bottom - dragBlurBounds.Height,
   dragBlurBounds.Y + point.Y - dragOrigin.Y));
  if (x == dragBlurBounds.X && y == dragBlurBounds.Y)
  {
   markStart.SetValue(movingBlur, originalBlurStart);
   markEnd.SetValue(movingBlur, originalBlurEnd);
   return;
  }
  markStart.SetValue(movingBlur, new Point(x, y));
  markEnd.SetValue(movingBlur, new Point(x + dragBlurBounds.Width, y + dragBlurBounds.Height));
 }

 public void EndBlurMove(bool cancel)
 {
  if (cancel)
  {
   markStart.SetValue(movingBlur, originalBlurStart);
   markEnd.SetValue(movingBlur, originalBlurEnd);
  }
  movingBlur = null;
  form.Capture = false;
  form.Cursor = Cursors.Cross;
 }

 public void UpdateBlurCursor(Point point)
 {
  if (FindBlur(point) != null) form.Cursor = Cursors.SizeAll;
 }

 public void InitializeBlur()
 {
  var mark = active.GetValue(form);
  if (mark == null || !IsBlur(mark)) return;
  markWidth.SetValue(mark, newBlurStrength);
  selectedBlur = mark;
 }

  }

 object FindBlur(Point point)
 {
  if ((string)currentTool.GetValue(form) != CaptureBlur.ToolName ||
      !(bool)selected.GetValue(form) || (int)mode.GetValue(form) != 0) return null;
  var area = (Rectangle)selection.GetValue(form);
  if (!area.Contains(point)) return null;
  var items = (IList)marks.GetValue(form);
  for (int i = items.Count - 1; i >= 0; i--)
  {
   var mark = items[i];
   if (!IsBlur(mark) || !BlurBounds(mark, area).Contains(point)) continue;
   foreach (var handle in (Point[])resizeHandles.Invoke(form, null))
    if (new Rectangle(handle.X - 7, handle.Y - 7, 14, 14).Contains(point)) return null;
   return mark;
  }
  return null;
 }

 public bool HasBlur
 {
  get
  {
   var current = active.GetValue(form);
   if (current != null && IsBlur(current)) return true;
   foreach (var mark in (IList)marks.GetValue(form))
    if (IsBlur(mark)) return true;

 public Bitmap RenderSelection()
 {
  if (!RequiresCustomRender) return (Bitmap)render.Invoke(form, null);
  var area = (Rectangle)selection.GetValue(form);
  try
  {
   RenderNative(false);
   return ReadFrame(area);
  }
  finally
  {
   ready = false;
   form.Invalidate();
  }
 }

 public bool RequiresCustomRender
 {
  get
  {
   var current = active.GetValue(form);
   if (current != null && (IsBlur(current) || IsMarker(current))) return true;
   foreach (var mark in (IList)marks.GetValue(form))
    if (IsBlur(mark) || IsMarker(mark)) return true;
   return false;
  }
 }

 bool IsBlur(object mark) { return (string)markTool.GetValue(mark) == CaptureBlur.ToolName; }
 bool IsMarker(object mark) { return (string)markTool.GetValue(mark) == CaptureMarker.ToolName; }

 Rectangle BlurBounds(object mark, Rectangle selectionArea)
 {
  var start = (Point)markStart.GetValue(mark);
  var end = (Point)markEnd.GetValue(mark);
  return Rectangle.Intersect(selectionArea, Rectangle.FromLTRB(
   Math.Min(start.X, end.X), Math.Min(start.Y, end.Y),
   Math.Max(start.X, end.X), Math.Max(start.Y, end.Y)));
 }

 void DrawMark(object mark, Rectangle selectionArea)
 {
  if (IsMarker(mark))
  {
   CaptureMarker.Draw(buffer, (Point)markStart.GetValue(mark), (List<Point>)markPoints.GetValue(mark),
    (int)markWidth.GetValue(mark));
   return;
  }
  if (!IsBlur(mark))
  {
   draw.Invoke(form, new object[] { buffer, mark });
   return;
  }
  var area = BlurBounds(mark, selectionArea);
  if (area.Width <= 0 || area.Height <= 0) return;
  buffer.Flush(FlushIntention.Sync);
  using (var source = ReadFrame(area))
   CaptureBlur.Draw(buffer, area, source, (int)markWidth.GetValue(mark));
 }

 void DrawBlurOutline(Rectangle area)
 {
  if (area.Width <= 0 || area.Height <= 0) return;
  using (var pen = new Pen(Color.FromArgb(143, 239, 180)))
   buffer.DrawRectangle(pen, area.X, area.Y, area.Width - 1, area.Height - 1);
 }

 Bitmap ReadFrame(Rectangle area)
 {
  var image = new Bitmap(area.Width, area.Height);
  try
  {
   using (var graphics = Graphics.FromImage(image))
   {
    var dc = graphics.GetHdc();
    try
    {
     if (!BitBlt(dc, 0, 0, area.Width, area.Height, frameSurface.Dc, area.X, area.Y, 0x00CC0020))
      throw new InvalidOperationException("Не удалось получить изображение с разметкой.");
    }
    finally { graphics.ReleaseHdc(dc); }
   }
   return image;
  }
  catch { image.Dispose(); throw; }
 }
